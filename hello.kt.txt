#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>
#include <algorithm>
using namespace std;

// 路线列表，共13条
vector<string> routes = {
    "A->B", "A->C", "A->D", "A->E",
    "B->A", "B->C", "B->D", "B->E",
    "C->A", "C->B", "C->D",
    "D->A", "D->B"
};

// 冲突对：两条路线不能同时放行
vector<pair<string, string>> conflictStr = {
    {"A->D", "E->B"},
    {"A->B", "B->D"},
    {"A->C", "C->B"},
    {"B->A", "D->B"},
    {"B->D", "C->A"},
    {"C->D", "D->A"},
    {"A->E", "B->C"},
    {"B->E", "C->A"},
    {"A->D", "C->B"}
};

// 将路线字符串映射成编号
unordered_map<string, int> name2id;
vector<vector<int>> adj; // 邻接表

// 贪心图着色：返回每个顶点的颜色(相位号)
vector<int> graphColoring(int n) {
    vector<int> color(n, -1);
    for (int u = 0; u < n; u++) {
        vector<bool> used(n, false);
        // 标记邻居已经使用的颜色
        for (int v : adj[u]) {
            if (color[v] != -1) {
                used[color[v]] = true;
            }
        }
        // 找最小可用颜色
        int c = 0;
        while (used[c]) c++;
        color[u] = c;
    }
    return color;
}

int main() {
    // 建立名字->id映射
    for (int i = 0; i < routes.size(); i++) {
        name2id[routes[i]] = i;
    }
    int n = routes.size();
    adj.resize(n);

    // 填充邻接表
    for (auto &p : conflictStr) {
        int u = name2id[p.first];
        int v = name2id[p.second];
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    vector<int> color = graphColoring(n);

    // 按相位分组
    vector<vector<string>> phases;
    for (int i = 0; i < n; i++) {
        int c = color[i];
        if (c >= phases.size()) phases.resize(c + 1);
        phases[c].push_back(routes[i]);
    }

    cout << "===== 五岔路口信号灯相位规划结果 =====\n";
    for (int i = 0; i < phases.size(); i++) {
        cout << "相位 " << i + 1 << " 放行路线：";
        for (auto &s : phases[i]) {
            cout << s << "  ";
        }
        cout << "\n";
    }
    cout << "\n总共有 " << phases.size() << " 个信号灯相位\n";
    return 0;
}
